Modify to support the settings-under-overlay case:
```ts
onHold: async () => {
  if (holdCommand) {
    await executeAction(holdCommand)
    return
  }
  const previousDeckId = getDisplayDeckId()
  // NEW: if the user is in the settings deck AND an overlay is active,
  // skip the back pop and show the overlay.
  if (previousDeckId === SETTINGS_DECK_ID && overlayDeckId !== null) {
    // Dismiss settings back to base, but since overlay is on top of base
    // and the user held-back, we need to also expose the overlay.
    // Concretely: pop settings, the overlay is already on top of base
    // (set up by the snapshot-driven push). So the user sees the overlay.
    deckController.goBack()
    await activateDeckSurface(overlayDeckId, previousDeckId)
    return
  }
  deckController.restoreStack([])
  await activateDeckSurface(undefined, previousDeckId)
}
```

The logic: when in settings, goBack goes to base. But the user wanted to go to the overlay. So after goBack, we explicitly `activateDeckSurface(overlayDeckId, ...)` which makes the overlay the visible deck.

This is one interpretation. An alternative: the overlay is always rendered on top of the active base deck (the runtime's render function already consults `overlayDeckId` separately from `deckController.getActiveDeckId()`). In that case, after goBack the user is on the base deck and the overlay is naturally on top — no explicit `activateDeckSurface(overlayDeckId, ...)` needed. Discover the actual semantics during execution and adjust.

In `packages/cli/src/addon/loader.ts`, find the function that registers addon decks (the place that walks `addon.decks` and adds them to the registry). After the walk completes, do a single pass over the registered decks:

```ts
function warnOnDuplicateProcessNames(
  decks: ReadonlyMap<string, AddonGeneratedDeck>,
  addonRegistry: AddonRegistry,
  logger: Logger,
): void {
  const seen = new Map<string, { addonName: string; deckId: string }>()
  for (const [key, deck] of decks) {
    if (!deck.process_names) continue
    for (const name of deck.process_names) {
      const normalized = name.toLowerCase().trim()
      if (!normalized) continue
      const existing = seen.get(normalized)
      if (existing) {
        logger.warn(
          {
            process_name: normalized,
            winner: existing.addonName,
            duplicate: addonRegistry.getDeckOwner(key),
          },
          `process '${normalized}' is declared by both '${existing.addonName}' and '${addonRegistry.getDeckOwner(key)}'; using '${existing.addonName}'`,
        )
      } else {
        seen.set(normalized, { addonName: addonRegistry.getDeckOwner(key) ?? 'unknown', deckId: key })
      }
    }
  }
}
```

Call this once after addon registration, passing the logger (which is already available in the loader's options).
