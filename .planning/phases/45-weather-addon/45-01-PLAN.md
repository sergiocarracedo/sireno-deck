  import type { HostContext } from '../../../system/host-context.js'
  import type { WeatherButtonConfig } from '../schemas.js'

  export interface WeatherSnapshot {
    available: boolean
    location: string
    temperature: number
    weatherCode: number
    windSpeed: number
    humidity: number
    source: string
  }

  export function createUnavailableWeatherSnapshot(source: string): WeatherSnapshot {
    return { available: false, location: '', temperature: 0, weatherCode: 0, windSpeed: 0, humidity: 0, source }
  }

  export function createWeatherController(options: {
    hostContext: HostContext
    config: WeatherButtonConfig
  }): WeatherController {
    return { async getSnapshot: () => fetchWeatherSnapshot(options.config) }
  }

  export interface WeatherController {
    getSnapshot: () => Promise<WeatherSnapshot>
  }

### 45-01-03: Add wttr.in fallback and ipapi.co geolocation

- **file**: `packages/cli/src/builtin-addons/weather/domain/wttr-in-fallback.ts` + `domain/ip-geolocation.ts`
- **action**: 

  `wttr-in-fallback.ts`:
  ```ts
  export async function fetchWttrInSnapshot(
    latitude: number,
    longitude: number,
    units: 'metric' | 'imperial',
  ): Promise<WeatherSnapshot> {
    const response = await fetch(`https://wttr.in/${latitude},${longitude}?format=j1`)
    if (!response.ok) throw new Error(`wttr.in ${response.status}`)
    const json = await response.json() as any
    const current = json.current_condition?.[0]
    if (!current) throw new Error('wttr.in: no current_condition')
    return {
      available: true,
      humidity: Number(current.humidity),
      location: `${current.observation_time ?? ''}`.trim() || 'wttr.in',
      source: 'wttr.in',
      temperature: Number(current.temp_C ?? current.temp_F),
      weatherCode: mapWttrCodeToWmo(current.weatherCode),
      windSpeed: Number(current.windspeedKmph ?? current.windspeedMiles),
    }
  }
  ```

  `ip-geolocation.ts`:
  ```ts
  export async function fetchIpGeolocation(): Promise<{ latitude: number; longitude: number; name: string } | null> {
    try {
      const r = await fetch('https://ipapi.co/json/')
      if (!r.ok) return null
      const j = await r.json() as any
      if (typeof j.latitude !== 'number' || typeof j.longitude !== 'number') return null
      return { latitude: j.latitude, longitude: j.longitude, name: j.city ?? j.country_name ?? 'IP location' }
    } catch { return null }
  }
  ```

### 45-01-05: Add the `weather` button

- **file**: `packages/cli/src/builtin-addons/weather/buttons/weather.tsx`
- **action**: 
  ```tsx
  import { defineMountedButton } from '@/addon/api'
  import { Surface } from '../components/Surface'
  import { createWeatherController, createUnavailableWeatherSnapshot, type WeatherSnapshot } from '../domain/weather-controller'
  import { WeatherButtonSchema } from '../schemas'

  type WeatherStoreState = { snapshot: WeatherSnapshot }

  function getState(snapshot: unknown): WeatherStoreState {
    return typeof snapshot === 'object' && snapshot !== null
      ? (snapshot as WeatherStoreState)
      : { snapshot: createUnavailableWeatherSnapshot('init') }
  }

  export const builtinWeatherButton = defineMountedButton({
    configSchema: WeatherButtonSchema,
    defaultIntervalMs: ({ config }) => config.render_interval_ms,
    defaultPollIntervalMs: ({ config }) => config.poll_interval_ms,
    onActivate: async ({ config, hostContext, store }) => {
      const controller = createWeatherController({ hostContext, config })
      try {
        const snap = await controller.getSnapshot()
        store.button.set({ snapshot: snap })
      } catch {
        store.button.set({ snapshot: createUnavailableWeatherSnapshot('init') })
      }
    },
    poll: async ({ config, hostContext, store }) => {
      const controller = createWeatherController({ hostContext, config })
      try {
        const snap = await controller.getSnapshot()
        store.button.set({ snapshot: snap })
      } catch { /* keep last */ }
    },
    render: ({ config, store }) => {
      const state = getState(store.button.snapshot)
      return <Surface snap={state.snapshot} unavailable={config.unavailable_label} />
    },
    type: 'weather',
  })
  ```
