# Addon API

TypeScript interfaces for creating addons.

## Core Types

### AddonManifestV1

```typescript
interface AddonManifestV1 {
  readonly apiVersion: 1
  readonly name: string
  readonly kind?: "runtime" | "theme"
  readonly buttonTypes: Readonly<Record<string, AddonButtonTypeDefAny>>
  readonly defaultButton?: string
  readonly decks?: ReadonlyArray<AddonDeckEntry>
  readonly frontend?: AddonFrontend
  readonly publishIntervalMs?: number
  readonly globalService?: AddonGlobalService
}
```

### AddonButtonTypeDef

```typescript
interface AddonButtonTypeDef<Config = unknown> {
  readonly frontend: AddonFrontendButton<Config>
  readonly service: AddonButtonTypeService<Config>
}

type AddonFrontendButton<Config> = ComponentType<
  AddonFrontendButtonProps<Config>
>

interface AddonFrontendButtonProps<Config> {
  readonly config: Config
  readonly state: unknown
  readonly addonName: string
  readonly buttonType: string
  readonly buttonId: string

}
```

### AddonGlobalService

```typescript
interface AddonGlobalService {
  readonly pollers?: ReadonlyArray<AddonGlobalPoller>

```

### AddonDeckEntry

```typescript
type AddonDeckEntry =
  | (AddonGeneratedDeck & { id: string })
  | { id: string; createDeck: (ctx: AddonDeckEntryCtx) => AddonGeneratedDeck }
  | {
      createDecks: (
        ctx: AddonDeckEntryCtx,
      ) => Record<string, AddonGeneratedDeck>
    }

interface AddonDeckEntryCtx {
  config: unknown
  deck: { id: string }
  keyCount: number
}

interface AddonGeneratedDeck {
  name?: string
  icon?: string
  background?: string
  buttonColor?: "blue" | "green" | "purple"
  buttons?: unknown[]
  paginated?: boolean
  trigger?: unknown
  autoShow?: boolean
  isOverlay?: boolean
}
```

## Store API

```typescript
interface Store {
  addonScope<T = unknown>(addonName: string): Scope<T>
  buttonScope<T = unknown>(addonName: string, buttonId: string): Scope<T>
  clearAddon(addonName: string): void
  clearButton(addonName: string, buttonId: string): void
  clearAll(): void
}

interface Scope<T = unknown> {
  get(key: string): T | undefined
  set(key: string, value: T): void
  update(key: string, fn: (current: T | undefined) => T): T
  clear(): void
  snapshot(): Readonly<Record<string, T>>
}
```

## Pub/Sub

```typescript
interface PubSub {
  publish<T extends Payload>(channel: string, payload: T): void
  subscribe<T extends Payload>(
    channel: string,
    cb: (payload: T) => void,
  ): Unsubscribe
  last<T extends Payload>(channel: string): T | undefined
}
```
