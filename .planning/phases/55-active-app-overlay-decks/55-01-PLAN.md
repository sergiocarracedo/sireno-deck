# Plan 55-01: System-flag foundation + active-app provider + monitor

<objective>
This plan lays the structural foundation for both the system-buttons/unified-dispatcher refactor and the active-app overlay work, without touching the runtime presentation layer. We:

1. Add `system?: boolean` to the `DeckConfig` interface and to `CoreDeckConfigSchema`. The user-facing bootstrap schema stays `.strict()` so a user can't smuggle `system: true` into YAML.
2. Stop using the `createImplicitSettingsDeck` / `createImplicitLockedDeck` factory pattern. Replace with a static `INTERNAL_DECKS: Record<string, DeckConfig>` const map (each entry with `system: true`) and merge it into `runtimeDecks` in the same loop as user decks — internal ids win (a user deck with `id: 'settings'` is shadowed).
3. Delete the broken `SystemOverlayToggleButton.tsx` and the duplicate `overlay-toggle-button.tsx`. Rewrite `system-buttons.ts` as a clean dispatcher that takes a single `systemContext` bundle and returns the right reserved-slot button (overlay-toggle / settings / back / null).
4. Add `process_names?: readonly string[]` to `AddonGeneratedDeck` (additive, `SIRENO_ADDON_API_VERSION` stays 1).
5. Create `packages/cli/src/system/active-app/` with an OS-abstracted `ActiveAppProvider` (`get-windows`-backed), an `unsupported` stub for pure Wayland and unknown OSes, a `getActiveAppProvider()` factory, and an `ActiveAppMonitor` class that polls at 500ms, dedupes, and emits changes. Install `get-windows` as a regular dependency.

Plan 55-02 wires the runtime overlay state and dismiss gestures on top of this foundation.
</objective>

```ts
import { logger as defaultLogger } from '@/util/logging'
import { createUnsupportedProvider } from './unsupported'
import type { ActiveAppProvider, ActiveAppSnapshot } from './provider'

function isPureWayland(): boolean {
  return (
    process.env.XDG_SESSION_TYPE === 'wayland' &&
    !process.env.WAYLAND_DISPLAY
  )
}

export async function createLinuxProvider(): Promise<ActiveAppProvider> {
  if (isPureWayland()) {
    return createUnsupportedProvider(defaultLogger)
  }
  // Lazy import so unsupported platforms don't load the native module
  const { activeWindow } = await import('get-windows')
  let stopped = false
  let timer: NodeJS.Timeout | null = null
  return {
    supportsActiveApp: true,
    start(onChange) {
      const tick = async () => {
        if (stopped) return
        try {
          const win = await activeWindow()
          const snap: ActiveAppSnapshot | null = win?.owner?.name
            ? { ownerName: win.owner.name }
            : null
          onChange(snap)
        } catch {
          onChange(null)
        }
        if (!stopped) timer = setTimeout(tick, 500)
      }
      void tick()
    },
    stop() {
      stopped = true
      if (timer) clearTimeout(timer)
    },
  }
}
```

```ts
}): ActiveAppMonitor {
  const interval = options.pollIntervalMs ?? 500
  let snapshot: ActiveAppSnapshot | null = null
  const listeners = new Set<(snap: ActiveAppSnapshot | null) => void>()
  let timer: NodeJS.Timeout | null = null

  function emit(next: ActiveAppSnapshot | null) {
    if (
      next?.ownerName === snapshot?.ownerName &&
      (next === null) === (snapshot === null)
    ) {
      return
    }
    snapshot = next
    for (const l of listeners) l(snapshot)
    options.onChange?.(snapshot)
  }

  return {
    start() {
      void options.provider.start(emit)
      // The provider runs its own setTimeout loop with `interval`.
      // The `interval` arg here is the cap if a provider wants to use it
      // via the helper in linux.ts/darwin.ts/windows.ts. We don't need a
      // second timer in the monitor itself.
    },
    stop() {
      void options.provider.stop()
      if (timer) clearTimeout(timer)
    },
    getSnapshot() { return snapshot },
    subscribe(l) { listeners.add(l); return () => { listeners.delete(l) } },
  }
}

export function createActiveAppMonitorDouble(
  initial: ActiveAppSnapshot | null = null,
): ActiveAppMonitor & { emit: (snap: ActiveAppSnapshot | null) => void } {
  const listeners = new Set<(snap: ActiveAppSnapshot | null) => void>()
  let snapshot = initial
  return {
    emit(snap) { snapshot = snap; for (const l of listeners) l(snap) },
    getSnapshot() { return snapshot },
    start() {},
    stop() {},
    subscribe(l) { listeners.add(l); return () => { listeners.delete(l) } },
  }
}
```
