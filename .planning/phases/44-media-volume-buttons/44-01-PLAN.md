  ```ts
  import type { HostContext } from '../../../system/host-context.js'

  export interface MediaVolumeSnapshot {
    available: boolean
    muted: boolean
    percentage: number
    source: string
  }

  export interface MediaVolumeController {
    getSnapshot: () => Promise<MediaVolumeSnapshot>
    setMuted: (muted: boolean) => Promise<boolean>
    setVolume: (deltaPercent: number) => Promise<boolean>
  }

  export function createUnavailableMediaVolumeSnapshot(source: string): MediaVolumeSnapshot {
    return { available: false, muted: false, percentage: 0, source }
  }

  export function createMediaVolumeController(options: {
    hostContext: HostContext
  }): MediaVolumeController {
    switch (options.hostContext.os.type) {
      case 'linux':
        return createLinuxMediaVolumeController()
      case 'macos':
        return createMacosMediaVolumeController()
      case 'windows':
        return createWindowsMediaVolumeController()
      default:
        return createUnsupportedMediaVolumeController(`${options.hostContext.os.type}-unsupported`)
    }
  }
  ```

### 44-01-02: Add Linux, macOS, and Windows adapter implementations

- **file**: `packages/cli/src/builtin-addons/media-player/domain/linux-media-volume-controller.ts`
- **action**: |
  Implement the Linux adapter using pactl against @DEFAULT_SINK@. Two files in fact:

  **linux-media-volume-controller.ts:**
  ```ts
  import { execa } from 'execa'
  import { createUnavailableMediaVolumeSnapshot, type MediaVolumeController, type MediaVolumeSnapshot } from './media-volume-controller.js'

  const SINK = '@DEFAULT_SINK@'
  const STEP = 5

  export function createLinuxMediaVolumeController(): MediaVolumeController {
    return {
      async getSnapshot(): Promise<MediaVolumeSnapshot> {
        try {
          const [muteResult, volumeResult] = await Promise.all([
            execa('pactl', ['get-sink-mute', SINK]),
            execa('pactl', ['get-sink-volume', SINK]),
          ])
          const muted = /yes/i.test(muteResult.stdout)
          const volumeMatch = /(\d+)%/.exec(volumeResult.stdout)
          const percentage = volumeMatch ? Math.min(100, Math.max(0, Number(volumeMatch[1]))) : 0
          return { available: true, muted, percentage, source: 'pactl' }
        } catch {
          return createUnavailableMediaVolumeSnapshot('pactl-unavailable')
        }
      },
      async setMuted(muted: boolean): Promise<boolean> {
        try {
          await execa('pactl', ['set-sink-mute', SINK, muted ? '1' : '0'])
          return true
        } catch { return false }
      },
      async setVolume(deltaPercent: number): Promise<boolean> {
        try {
          const sign = deltaPercent >= 0 ? '+' : ''
          await execa('pactl', ['set-sink-volume', SINK, `${sign}${deltaPercent}%`])
          return true
        } catch { return false }
      },
    }
  }
  ```

  **macos-media-volume-controller.ts:** mirror using osascript:
  - `getSnapshot`: `osascript -e 'output muted of (get volume settings)'` and parse `output volume`
  - `setMuted`: `osascript -e 'set volume output muted to (true|false)'`
  - `setVolume`: compute new value 0-100 and `osascript -e 'set volume output volume N'`

  **windows-media-volume-controller.ts:** all methods return `createUnavailableMediaVolumeController(...).createUnavailableSnapshot('windows-unsupported')` — the platform stays unsupported in v1.4.

  **Unsupported controller:** `createUnavailableMediaVolumeController(source)` returns a controller that always returns `createUnavailableMediaVolumeSnapshot(source)`, `setMuted`/`setVolume` return `false`.
- **verify**: `cd packages/cli && npx tsc --noEmit -p . 2>&1 | head -10`
- **done**: All four adapters (linux, macos, windows, unsupported) implemented.

### 44-01-04: Add the `media-mute` button

- **file**: `packages/cli/src/builtin-addons/media-player/buttons/media-mute.tsx`
- **action**: |
  Implement the mute toggle button:

  ```tsx
  import { useEffect, useState } from 'react'
  import { ButtonSurface, defineMountedButton } from '../../../addon/api.js'
  import { Icon, Text } from '../../../ui/index.js'
  import { createMediaVolumeController, type MediaVolumeSnapshot } from '../domain/media-volume-controller.js'
  import { BuiltinMediaMuteButtonSchema } from '../schemas.js'
  import { createUnavailableMediaVolumeSnapshot } from '../domain/media-volume-controller.js'

  const POLL_MS = 2_500

  function renderMuteSurface(snap: MediaVolumeSnapshot) {
    return (
      <ButtonSurface full>
        <div className="flex h-full w-full flex-col items-center justify-center gap-1">
          <Icon
            icon={snap.muted ? 'volume-x' : 'volume-2'}
            size={28}
            tone={snap.muted ? 'danger' : 'foreground'}
          />
          <Text size="xs" tone={snap.muted ? 'danger' : 'foreground'}>
            {snap.muted ? 'Muted' : 'Audio'}
          </Text>
        </div>
      </ButtonSurface>
    )
  }

  export const builtinMediaMuteButton = defineMountedButton({
    configSchema: BuiltinMediaMuteButtonSchema,
    defaultPollIntervalMs: POLL_MS,
    onActivate: async ({ hostContext, store }) => {
      const controller = createMediaVolumeController({ hostContext })
      store.button.set({ snapshot: createUnavailableMediaVolumeSnapshot('init') })
      try {
        const snap = await controller.getSnapshot()
        store.button.set({ snapshot: snap })
      } catch { /* keep unavailable */ }
    },
    onTap: async ({ hostContext, methods, store }) => {
      const state = (store.button.snapshot as { snapshot?: MediaVolumeSnapshot } | undefined)?.snapshot
      const controller = createMediaVolumeController({ hostContext })
      const next = !(state?.muted ?? false)
      const ok = await controller.setMuted(next)
      if (ok) {
        const snap = await controller.getSnapshot()
        store.button.set({ snapshot: snap })
      }
      methods.invalidate()
    },
    render: ({ store }) => {
      const state = (store.button.snapshot as { snapshot?: MediaVolumeSnapshot } | undefined)
      return renderMuteSurface(state?.snapshot ?? createUnavailableMediaVolumeSnapshot('init'))
    },
    type: 'media-mute',
  })
  ```

  Note: the runtime re-invokes render() on defaultPollIntervalMs ticks (which also drives the snapshot refresh via the poll function). The render reads the latest snapshot from the store.
- **verify**: `cd packages/cli && npx tsc --noEmit -p . 2>&1 | head -10`
- **done**: `media-mute` button file compiles; icon swaps between volume-2 and volume-x based on muted state.

### 44-01-05: Add the `media-volume` button with up/down variants and hold-to-mute

- **file**: `packages/cli/src/builtin-addons/media-player/buttons/media-volume.tsx`
- **action**: |
  Implement the volume up/down button:

  ```tsx
  import { useEffect, useState } from 'react'
  import { ButtonSurface, defineMountedButton } from '../../../addon/api.js'
  import { Bars, Icon, Text } from '../../../ui/index.js'
  import { createMediaVolumeController, type MediaVolumeSnapshot } from '../domain/media-volume-controller.js'
  import { BuiltinMediaVolumeButtonSchema } from '../schemas.js'
  import { createUnavailableMediaVolumeSnapshot } from '../domain/media-volume-controller.js'

  const POLL_MS = 1_500
  const STEP = 5
  const HOLD_MS = 600

  function renderVolumeSurface(variant: 'up' | 'down', snap: MediaVolumeSnapshot) {
    const arrow = variant === 'up' ? '▲' : '▼'
    return (
      <ButtonSurface full>
        <div className="flex h-full w-full flex-col items-center justify-center gap-1">
          <Icon
            icon={variant === 'up' ? 'volume-1' : 'volume-1'}
            size={22}
            tone="foreground"
          />
          <Text size="md" tone="primary">
            {arrow} {snap.percentage}%
          </Text>
          <Bars value={snap.percentage} max={100} className="w-3/4" />
        </div>
      </ButtonSurface>
    )
  }

  export const builtinMediaVolumeButton = defineMountedButton({
    configSchema: BuiltinMediaVolumeButtonSchema,
    defaultPollIntervalMs: POLL_MS,
    onActivate: async ({ config, hostContext, store }) => {
      const controller = createMediaVolumeController({ hostContext })
      const snap = await controller.getSnapshot()
      store.button.set({ snapshot: snap })
    },
    onTap: async ({ config, hostContext, methods, store }) => {
      const controller = createMediaVolumeController({ hostContext })
      const delta = config.variant === 'up' ? STEP : -STEP
      await controller.setVolume(delta)
      const snap = await controller.getSnapshot()
      store.button.set({ snapshot: snap })
      methods.invalidate()
    },
    onPress: ({ store }) => {
      store.button.set({ holdTimer: setTimeout(() => { /* mute toggle via hold */ }, HOLD_MS) })
    },
    onRelease: ({ config, hostContext, methods, store }) => {
      const state = (store.button.snapshot as { holdTimer?: ReturnType<typeof setTimeout> } | undefined)
      if (state?.holdTimer) clearTimeout(state.holdTimer)
      // Hold-to-mute: if held past threshold, toggle mute
      const held = (store.button.snapshot as { holdStartedAt?: number } | undefined)?.holdStartedAt
      if (held && Date.now() - held >= HOLD_MS) {
        const controller = createMediaVolumeController({ hostContext })
        void controller.getMuted().then(async (muted) => {
          await controller.setMuted(!muted)
          const snap = await controller.getSnapshot()
          store.button.set({ snapshot: snap })
          methods.invalidate()
        })
      }
    },
    render: ({ config, store }) => {
      const state = (store.button.snapshot as { snapshot?: MediaVolumeSnapshot } | undefined)
      return renderVolumeSurface(config.variant, state?.snapshot ?? createUnavailableMediaVolumeSnapshot('init'))
    },
    type: 'media-volume',
  })
  ```

  Note: the hold-to-mute logic in onRelease checks if the press duration exceeded HOLD_MS. The poll-driven render() then refreshes the snapshot.
- **verify**: `cd packages/cli && npx tsc --noEmit -p . 2>&1 | head -10`
- **done**: `media-volume` button file compiles with up/down variants; hold-to-mute logic present.
