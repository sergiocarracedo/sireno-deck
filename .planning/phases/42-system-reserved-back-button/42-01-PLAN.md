  4. Error message: `Button at reserved slot (position N) in deck "<id>" cannot be claimed by addons. Reserved for the system back button. Set "allow_reserved_slot_override: true" on the deck (or root) to override.`

  ```ts
  .refine(
    (config) => {
      if (config.allow_reserved_slot_override) return true
      const lockedDeckId = config.session?.locked_deck
      for (const deck of Object.values(config.decks ?? {})) {
        if (deck.id === lockedDeckId) continue
        if (deck.allow_reserved_slot_override) continue
        const reserved = (deck.buttons?.length ?? 0) > 0 ? deck.keyCount - 1 : -1
        if (deck.buttons?.some((b: any) => b.position === reserved)) {
          return false
        }
      }
      return true
    },
    { message: 'Button at reserved slot cannot be claimed by addons. Use "allow_reserved_slot_override: true" to override.' },
  )
  ```

### 42-01-03: Add SystemBackButton component

- **file**: `packages/cli/src/deck/system-back-button.tsx`
- **action**: |
  Create the system back button / home indicator component. Single file with two visual variants:

  ```tsx
  import { ButtonSurface, Text } from '@sireno-deck-cli'
  import { Icon } from '@/ui'
  import { ReactElement, useRef } from 'react'

  const HOLD_THRESHOLD_MS = 600

  interface SystemBackButtonProps {
    isMainDeck: boolean
    onTap: () => void
    onHold: () => void
    backIconOverride?: string // optional theme override
  }

  export function SystemBackButton(props: SystemBackButtonProps): ReactElement {
    const { isMainDeck, onTap, onHold, backIconOverride } = props
    const holdTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
    const holdTriggeredRef = useRef(false)

    if (isMainDeck) {
      return (
        <ButtonSurface>
          <div className="flex h-full w-full items-center justify-center">
            <Text size="xs" tone="foreground" className="opacity-30">
              Home
            </Text>
          </div>
        </ButtonSurface>
      )
    }

    const onPointerDown = () => {
      holdTriggeredRef.current = false
      holdTimerRef.current = setTimeout(() => {
        holdTriggeredRef.current = true
        onHold()
      }, HOLD_THRESHOLD_MS)
    }
    const onPointerUp = () => {
      if (holdTimerRef.current) clearTimeout(holdTimerRef.current)
      if (!holdTriggeredRef.current) onTap()
    }

    return (
      <ButtonSurface>
        <button
          className="flex h-full w-full items-center justify-center gap-1"
          onPointerDown={onPointerDown}
          onPointerUp={onPointerUp}
        >
          <Icon icon={backIconOverride ?? 'chevron-left'} size={16} />
          <Text size="sm" tone="foreground">Back</Text>
        </button>
      </ButtonSurface>
    )
  }
  ```
- **verify**: File exists, exports `SystemBackButton`. TypeScript compiles.
- **done**: Component created with both isMainDeck and back variants, hold timer reuses 600ms threshold.
