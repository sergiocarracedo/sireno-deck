# Plan 54-02: Core settings deck with brightness up/down

<objective>
After this plan, the runtime has a core-managed `settings` deck. The deck renders 6 buttons: brightness-up (+), brightness-down (−), a current-brightness label (e.g. "75%"), the project logo + CLI version (via the `<LogoVersion />` element shipped in 54-01), and two visual-balance buttons. The reserved slot is the standard back button (handled by the system-back-button injection). The settings deck is the home of the logo+version (per the v1.5 prompt: the logo+version "is in the main deck now" — actually, the user clarified in the discussion that the settings deck IS the new home, and the main deck's reserved slot becomes a settings affordance).

```tsx
import type { ReactElement } from "react"

import { setBrightnessAll, getCurrentBrightness } from "@/device/registry"
import { LogoVersion } from "@/ui/LogoVersion"
import { Text } from "@/ui/Text"
import { ButtonSurface } from "@/addon/api"

const STEP = 10

export function nextBrightnessUp(current: number): number {
  return Math.min(100, current + STEP)
}

export function nextBrightnessDown(current: number): number {
  return Math.max(0, current - STEP)
}

export interface SettingsDeckButtonProps {
  onTap: () => void
}

export function BrightnessUpButton(_props: SettingsDeckButtonProps): ReactElement {
  return (
    <ButtonSurface full>
      <button
        className="flex h-full w-full flex-col items-center justify-center gap-1"
        data-sireno-brightness-up="true"
        onClick={_props.onTap}
        type="button"
      >
        <Text size="3xl" tone="primary">+</Text>
        <Text size="xs" tone="foreground">Brightness</Text>
      </button>
    </ButtonSurface>
  )
}

export function BrightnessDownButton(_props: SettingsDeckButtonProps): ReactElement {
  return (
    <ButtonSurface full>
      <button
        className="flex h-full w-full flex-col items-center justify-center gap-1"
        data-sireno-brightness-down="true"
        onClick={_props.onTap}
        type="button"
      >
        <Text size="3xl" tone="primary">−</Text>
        <Text size="xs" tone="foreground">Brightness</Text>
      </button>
    </ButtonSurface>
  )
}

export function CurrentBrightnessLabel(): ReactElement {
  return (
    <ButtonSurface full>
      <div
        className="flex h-full w-full flex-col items-center justify-center gap-0.5"
        data-sireno-current-brightness="true"
      >
        <Text size="2xl" tone="primary">{getCurrentBrightness()}%</Text>
        <Text size="xs" tone="foreground">Current</Text>
      </div>
    </ButtonSurface>
  )
}

export function renderSettingsDeck(): ReactElement {
  const current = getCurrentBrightness()
  return (
    <div className="grid h-full w-full gap-1" data-sireno-settings-deck="true">
      <BrightnessUpButton
        onTap={async () => {
          await setBrightnessAll(nextBrightnessUp(current))
        }}
      />
      <BrightnessDownButton
        onTap={async () => {
          await setBrightnessAll(nextBrightnessDown(current))
        }}
      />
      <CurrentBrightnessLabel />
      <LogoVersion />
    </div>
  )
}
```
