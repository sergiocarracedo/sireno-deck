# Plan 55-01: process_names schema + OS-abstracted active-app provider + monitor

<objective>
This plan lays the foundation for active-app overlay decks without touching the runtime. We add the `process_names?: string[]` field to the `AddonGeneratedDeck` addon API, install `get-windows` as a regular dependency, and create the `packages/cli/src/system/active-app/` module that abstracts the per-platform foreground-window polling behind a single `ActiveAppProvider` interface. The `getActiveAppProvider()` factory selects the right provider (or returns an `unsupported` stub on pure Wayland / unknown OS). The `ActiveAppMonitor` class wraps any provider, polls at 500ms, dedupes by owner name, and emits change events via an `onChange` callback. The plan is verifiable on its own: the unit tests confirm the provider selection, the Wayland-detection, and the monitor's dedup-and-emit behavior.

The runtime integration (subscribing to the monitor, pushing/popping overlay decks, dismiss gestures) is plan 55-02. The settings-under-overlay layering and duplicate-warning at addon load time are plan 55-03.
</objective>

Create `packages/cli/src/system/active-app/linux.ts`:
```ts
import { createUnsupportedProvider } from './unsupported'
import type {
  ActiveAppProvider,
  ActiveAppProviderDeps,
} from './provider'

const POLL_INTERVAL_MS = 500

export function createLinuxProvider(
  deps: ActiveAppProviderDeps,
): ActiveAppProvider {
  if (
    process.env.XDG_SESSION_TYPE === 'wayland' &&
    !process.env.WAYLAND_DISPLAY
  ) {
    return createUnsupportedProvider(deps, 'pure-wayland')
  }

  let timer: ReturnType<typeof setTimeout> | undefined
  let lastName: string | null = null
  return {
    supportsActiveApp: true,
    start(onChange) {
      void poll()
      async function poll(): Promise<void> {
        try {
          const { activeWindow } = await import('get-windows')
          const win = await activeWindow()
          const name = win?.owner?.name ?? null
          if (name !== lastName) {
            lastName = name
            onChange(name === null ? null : { ownerName: name })
          }
        } catch (error) {
          deps.logger.warn({ error }, 'active-app: linux poll failed')
        }
        timer = setTimeout(poll, POLL_INTERVAL_MS)
      }
    },
    stop() {
      if (timer) clearTimeout(timer)
    },
  }
}
```

import { createLinuxProvider } from './linux'
import { createUnsupportedProvider } from './unsupported'
import { createWindowsProvider } from './windows'
import type {
  ActiveAppProvider,
  ActiveAppProviderDeps,
} from './provider'

export type { ActiveAppProvider, ActiveAppSnapshot } from './provider'

export interface GetActiveAppProviderOptions extends ActiveAppProviderDeps {
  platform?: NodeJS.Platform
}

export function getActiveAppProvider(
  opts: GetActiveAppProviderOptions,
): ActiveAppProvider {
  const platform = opts.platform ?? process.platform
  const deps: ActiveAppProviderDeps = { logger: opts.logger }
  switch (platform) {
    case 'linux':
      return createLinuxProvider(deps)
    case 'darwin':
      return createDarwinProvider(deps)
    case 'win32':
      return createWindowsProvider(deps)
    default:
      return createUnsupportedProvider(deps, `unknown-platform:${platform}`)
  }
}
