# Plan 50-02: 2-day daily forecast page

<objective>
The weather button gets a new fourth page that renders the next 2 days as side-by-side columns: short day label, weather icon, high temperature, low temperature, and precipitation sum. The existing hourly forecast page is renamed to `hourly-forecast` for self-documentation but keeps the same visual layout. The daily data is fetched in the SAME HTTP call as the current weather — the request gains a `daily=` parameter and the response parser reads the new `daily` block.
</objective>

```ts
function buildDailyEntries(
  daily: OpenMeteoDaily | undefined,
): DailyForecastEntry[] {
  if (!daily?.time?.length) return []
  const out: DailyForecastEntry[] = []
  for (let i = 0; i < 2 && i < daily.time.length; i++) {
    const date = daily.time[i]
    if (date === undefined) break
    out.push({
      date,
      weatherCode: daily.weather_code[i] ?? 0,
      tempMax: daily.temperature_2m_max[i] ?? 0,
      tempMin: daily.temperature_2m_min[i] ?? 0,
      precipitationSum: daily.precipitation_sum[i] ?? 0,
    })
  }
  return out
}
```

Create `packages/cli/src/builtin-addons/weather/buttons/components/DailyForecast.tsx`:

```tsx
import type { ReactElement } from 'react'

import { Text } from '@/ui'

import type { DailyForecastEntry } from '../../domain/weather-controller'
import { convertTemperature } from '../../domain/unit-conversion'
import { WmoIcon } from './WmoIcon'

const DAY_LABELS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']

function formatDayLabel(isoDate: string): string {
  const d = new globalThis.Date(`${isoDate}T12:00:00Z`)
  if (Number.isNaN(d.getTime())) return isoDate
  return DAY_LABELS[d.getUTCDay()] ?? isoDate
}

export function DailyForecast({
  entries,
  units,
}: {
  entries: DailyForecastEntry[]
  units: 'metric' | 'imperial'
}): ReactElement {
  if (entries.length === 0) {
    return (
      <div className="flex h-full w-full items-center justify-center">
        <Text size="xs" tone="foreground">
          No daily forecast
        </Text>
      </div>
    )
  }
  return (
    <div className="flex h-full w-full items-stretch justify-between gap-0.5 px-1">
      {entries.map((entry) => {
        const max = convertTemperature(entry.tempMax, units)
        const min = convertTemperature(entry.tempMin, units)
        return (
          <div
            className="flex min-w-0 flex-1 flex-col items-center justify-center gap-0.5"
            key={entry.date}
          >
            <Text size="xs" tone="primary">
              {formatDayLabel(entry.date)}
            </Text>
            <WmoIcon code={entry.weatherCode} size={14} />
            <Text size="xs" tone="primary">
              {max.value}
              {max.units}
            </Text>
            <Text size="xs" tone="foreground">
              {min.value}
              {min.units}
            </Text>
            <Text size="xs" tone="foreground">
              {entry.precipitationSum}mm
            </Text>
          </div>
        )
      })}
    </div>
  )
}
```
