### 46-01-03: Add pagination utility function

- **file**: `packages/cli/src/builtin-addons/emoji-selector/support.tsx`
- **action**: Add a `paginateEmojis` function:
  ```ts
  export interface EmojiPage {
    emojis: string[]
    pageIndex: number
    totalPages: number
  }

  export function paginateEmojis(emojis: readonly string[], pageSize: number): EmojiPage[] {
    if (emojis.length === 0) return []

    const firstPageSize = pageSize
    const restPageSize = pageSize - 1 // one slot occupied by prev button

    const pages: EmojiPage[] = []
    let remaining = [...emojis]

    // First page
    pages.push({
      emojis: remaining.slice(0, firstPageSize),
      pageIndex: 0,
      totalPages: 0, // will compute after
    })
    remaining = remaining.slice(firstPageSize)

    // Subsequent pages
    while (remaining.length > 0) {
      pages.push({
        emojis: remaining.slice(0, restPageSize),
        pageIndex: 0,
        totalPages: 0,
      })
      remaining = remaining.slice(restPageSize)
    }

    // Back-fill totalPages
    const total = pages.length
    for (const page of pages) {
      page.totalPages = total
    }

    return pages
  }
  ```
- **verify**: `cd packages/cli && npx tsc --noEmit -p . 2>&1 | head -10`
- **done**: Function exported with correct typing.
