```ts
import { readFileSync, writeFileSync, unlinkSync, existsSync, mkdirSync } from "node:fs"
import { homedir, platform } from "node:os"
import { join } from "node:path"
import type pino from "pino"

function getPidDir(): string {
  const base = platform() === "linux"
    ? join(homedir(), ".local", "state")
    : join(homedir(), ".local", "share")
  return join(base, "sireno-deck")
}

function getPidPath(): string {
  return join(getPidDir(), "daemon.pid")
}

export function ensurePidDir(): void {
  mkdirSync(getPidDir(), { recursive: true })
}

export function writePid(): void {
  ensurePidDir()
  writeFileSync(getPidPath(), String(process.pid), "utf-8")
}

export function readPid(): number | null {
  try {
    const content = readFileSync(getPidPath(), "utf-8").trim()
    return Number.parseInt(content, 10)
  } catch {
    return null
  }
}

export function isRunning(pid: number): boolean {
  try {
    return process.kill(pid, 0)
  } catch {
    return false
  }
}

export function removePidFile(): void {
  try {
    unlinkSync(getPidPath())
  } catch {
    // File may already be gone — that's fine
  }
}

export function setupSignalHandlers(logger: pino.Logger): () => void {
  const cleanup = () => {
    logger.info("shutting down...")
    // Phase 2 will add device disconnect here
    removePidFile()
    logger.flush()
    process.exit(0)
  }

  process.on("SIGTERM", cleanup)
  process.on("SIGINT", cleanup)

  return () => {
    process.off("SIGTERM", cleanup)
    process.off("SIGINT", cleanup)
  }
}
```

- readPid returns null when no PID file exists

import { loadConfig } from "../../config/loader.js"
import { formatConfigError } from "../../util/errors.js"
import { ConfigValidationError } from "../../core/schemas.js"
import { writePid, readPid, isRunning, removePidFile, setupSignalHandlers } from "../../util/daemon.js"
import type pino from "pino"

export interface StartOptions {
  config?: string
  logger: pino.Logger
}

export async function startDaemon(options: StartOptions): Promise<void> {
  const { logger } = options

  // Check for existing running daemon
  const existingPid = readPid()
  if (existingPid !== null && isRunning(existingPid)) {
    logger.error({ pid: existingPid }, "daemon already running (PID: %d)", existingPid)
    process.exit(1)
  }

  // Remove stale PID file
  removePidFile()

  // Load config
  try {
    const config = loadConfig(options.config)
    logger.info({ config }, "config loaded successfully")
  } catch (err) {
    if (err instanceof ConfigValidationError) {
      console.error(formatConfigError(err))
      process.exit(1)
    }
    throw err
  }

  // Write PID and register signal handlers
  writePid()
  setupSignalHandlers(logger)

  logger.info("sireno-deck daemon started (PID: %d)", process.pid)
  logger.info("press Ctrl+C to stop")

  // Keep the process alive
  await new Promise(() => {})
}
```

import { readPid, isRunning, removePidFile } from "../../util/daemon.js"
import type pino from "pino"

export interface StopOptions {
  logger: pino.Logger
}

export async function stopDaemon(options: StopOptions): Promise<void> {
  const { logger } = options
  const pid = readPid()

  if (pid === null) {
    logger.info("no running daemon found")
    return
  }

  if (!isRunning(pid)) {
    logger.warn({ pid }, "stale PID file found (PID %d not running)", pid)
    removePidFile()
    return
  }

  logger.info({ pid }, "stopping daemon (PID: %d)", pid)
  process.kill(pid, "SIGTERM")

  // Wait for process to exit
  let retries = 10
  while (retries > 0 && isRunning(pid)) {
    await new Promise((r) => setTimeout(r, 100))
    retries--
  }

  if (isRunning(pid)) {
    logger.warn({ pid }, "daemon did not respond to SIGTERM, sending SIGKILL")
    process.kill(pid, "SIGKILL")
  }

  logger.info("daemon stopped")
}
```

export interface StatusOptions {
  logger: pino.Logger
}

export async function checkStatus(options: StatusOptions): Promise<void> {
  const { logger } = options
  const pid = readPid()

  if (pid === null) {
    logger.info("daemon is not running")
    return
  }

  if (isRunning(pid)) {
    logger.info({ pid }, "daemon is running (PID: %d)", pid)
    return
  }

  logger.warn({ pid }, "stale PID file found (PID %d not running)", pid)
  removePidFile()
}
